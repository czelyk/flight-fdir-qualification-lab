cmake_minimum_required(VERSION 3.20)
project(flight_fdir_qualification_lab VERSION 0.5.0 LANGUAGES C)
if(ZIG_EXECUTABLE)
  set(CMAKE_C_ARCHIVE_CREATE "<CMAKE_AR> ar qc <TARGET> <LINK_FLAGS> <OBJECTS>")
  set(CMAKE_C_ARCHIVE_APPEND "<CMAKE_AR> ar q <TARGET> <LINK_FLAGS> <OBJECTS>")
  set(CMAKE_C_ARCHIVE_FINISH "")
endif()
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)
add_library(fq_monitor STATIC src/fq_monitor.c src/fq_tickless.c src/fq_acquisition.c src/fq_ingress.c src/fq_handoff.c)
set_target_properties(fq_monitor PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(fq_monitor PUBLIC include)
add_library(fq_bridge SHARED tests/bridge.c)
target_link_libraries(fq_bridge PRIVATE fq_monitor)
add_library(fq_worker_example OBJECT examples/worker.c)
target_link_libraries(fq_worker_example PRIVATE fq_monitor)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(fq_monitor PRIVATE -Wall -Wextra -Werror)
  target_compile_options(fq_bridge PRIVATE -Wall -Wextra -Werror)
  target_compile_options(fq_worker_example PRIVATE -Wall -Wextra -Werror)
endif()
if(BUILD_TESTING)
  add_executable(fq_unit tests/unit.c)
  target_link_libraries(fq_unit PRIVATE fq_monitor)
  add_test(NAME c_boundary_unit_tests COMMAND fq_unit)
  add_executable(fq_tickless_checks tests/tickless.c)
  target_link_libraries(fq_tickless_checks PRIVATE fq_monitor)
  add_test(NAME c_tickless_equivalence COMMAND fq_tickless_checks)
  add_executable(fq_acquisition_checks tests/acquisition.c)
  target_link_libraries(fq_acquisition_checks PRIVATE fq_monitor)
  add_test(NAME c_first_sample_watchdog COMMAND fq_acquisition_checks)
  add_executable(fq_ingress_checks experiments/generation_queue.c)
  target_link_libraries(fq_ingress_checks PRIVATE fq_monitor)
  add_test(NAME c_generation_queue_comparison COMMAND fq_ingress_checks)
  add_executable(fq_handoff_checks experiments/handoff_order.c)
  target_link_libraries(fq_handoff_checks PRIVATE fq_monitor)
  add_test(NAME c_delayed_epoch_handoff COMMAND fq_handoff_checks)
endif()
